(() => {
  const params = new URLSearchParams(location.search);
  const paths = params.getAll("p");

      const cs = win.getComputedStyle(p);
      if (cs.overflowX !== "visible" || cs.overflow !== "visible") return true;
      p = p.parentElement;
    }
    return false;
  }

    const doc = frame.contentDocument;

    });
    return {
      path, w, vw,
      docSW: doc.documentElement.scrollWidth,
      bodySW: doc.body.scrollWidth,
      overflow: doc.documentElement.scrollWidth - vw,
      offenders: bad.slice(0, 6),
      extra: Math.max(0, bad.length - 6)
    };

      const frame = document.createElement("iframe");
      frame.setAttribute("width", String(entry.w));
      frame.style.cssText = `width:${entry.w}px;height:900px;border:0;position:absolute;left:-10000px;top:0`;
      frame.src = entry.path;
      const finish = () => {
        setTimeout(() => {
          try { rows.push(scan(frame, entry.path, entry.w)); }
          catch (e) { rows.push({ path: entry.path, w: entry.w, error: String(e) }); }
          frame.remove();
          done += 1;
          report.textContent = `PROGRESS ${done}/${total}\n` + rows.map(render).join("\n");
          resolve();
        }, 350);
      };
      frame.addEventListener("load", finish, { once: true });
      document.body.appendChild(frame);
    });
  }

  function render(r) {
    if (r.error) return `! ${r.path} @${r.w} -> ${r.error}`;
    const flag = r.overflow > 0 ? "OVERFLOW" : "ok";
    return `[${flag}] ${r.path} @${r.w} vw=${r.vw} docScrollW=${r.docSW} bodyScrollW=${r.bodySW} delta=${r.overflow}` +
      (r.offenders.length ? `\n      offenders: ${r.offenders.join(" | ")}${r.extra ? ` (+${r.extra} more)` : ""}` : "");
  }

  const queue = [];
  for (const p of paths) for (const w of widths) queue.push({ path: p, w });

  const workers = Array.from({ length: 4 }, async () => {
    while (queue.length) {
      const next = queue.shift();
      if (!next) break;
      await run(next);
    }
  });

  Promise.all(workers).then(() => {
    report.textContent = "DONE\n" + render({ path: "", w: 0, vw: 0, docSW: 0, bodySW: 0, overflow: 0 }) + "\n";
    report.textContent = "DONE total=" + total + " overflowCount=" + rows.filter((r) => r.overflow > 0).length + "\n" +
      rows.map(render).join("\n");
    document.title = "AUDIT-DONE";
  });
})();</script>
